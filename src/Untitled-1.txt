import { useEffect, useMemo, useState } from 'react'
import { supabase } from './supabase'
import './CustomerPage.css'

type Product = {
  id: number
  name: string
  category: string
  price: number
  stock: number
  active?: boolean
  available?: boolean
}

type CartItem = Product & {
  quantity: number
}

type Coupon = {
  code: string
  percent: number
  active?: boolean
}

type Notice = {
  message: string
  type: 'success' | 'error' | 'info'
}

type Confirmation = {
  orderId: number
  total: number
  type: 'سفري' | 'داخل الكوفي'
  table: string
}

const normalizeProduct = (item: any): Product => {
  const stock = Number(item?.stock ?? 0)
  const price = Number(item?.price ?? 0)
  const isActive = item?.active ?? true

  return {
    id: Number(item?.id ?? 0),
    name: String(item?.name ?? 'منتج غير مسمى'),
    category: String(item?.category ?? 'عام'),
    price,
    stock,
    active: Boolean(isActive),
    available: Boolean(item?.available ?? (isActive && stock > 0)),
  }
}

const normalizeCoupon = (item: any): Coupon => ({
  code: String(item?.code ?? '').trim().toUpperCase(),
  percent: Number(item?.percent ?? 0),
  active:
    item?.active === undefined
      ? true
      : Boolean(item.active),
})

export default function CustomerPage() {
  const [products, setProducts] = useState<Product[]>([])
  const [coupons, setCoupons] = useState<Coupon[]>([])
  const [selectedCategory, setSelectedCategory] = useState('الكل')
  const [selectedProduct, setSelectedProduct] = useState<Product | null>(null)
  const [cart, setCart] = useState<CartItem[]>([])
  const [orderType, setOrderType] = useState<'سفري' | 'داخل الكوفي' | ''>('')
  const [tableNumber, setTableNumber] = useState('')
  const [phoneNumber, setPhoneNumber] = useState('')
  const [paymentMethod, setPaymentMethod] = useState<'نقدي' | 'بطاقة'>('نقدي')
  const [discountCode, setDiscountCode] = useState('')
  const [discountPercent, setDiscountPercent] = useState(0)
  const [notice, setNotice] = useState<Notice | null>(null)
  const [isSubmitting, setIsSubmitting] = useState(false)
  const [confirmation, setConfirmation] = useState<Confirmation | null>(null)
  const [showSplash, setShowSplash] = useState(true)
  const [showOrderChoice, setShowOrderChoice] = useState(true)
  const [customerScreen, setCustomerScreen] = useState<'menu' | 'cart'>('menu')

  useEffect(() => {
    const timer = window.setTimeout(() => {
      setShowSplash(false)
    }, 2300)

    return () => window.clearTimeout(timer)
  }, [])

  const notify = (
    message: string,
    type: Notice['type'] = 'info',
  ) => {
    setNotice({ message, type })
    window.setTimeout(() => setNotice(null), 3200)
  }

  useEffect(() => {
    const loadProducts = async () => {
      try {
        const { data, error } = await supabase
          .from('products')
          .select('*')
          .order('id', { ascending: true })

        if (error) {
          console.error('Customer products error:', error)
          return
        }

        const dbProducts = (data ?? [])
          .map(normalizeProduct)
          .filter((product) => product.active !== false)

        setProducts(dbProducts)
        if (!selectedProduct && dbProducts.length > 0) {
          setSelectedProduct(dbProducts[0])
        }
      } catch (error) {
        console.error('Customer products load failed:', error)
      }
    }

    const loadCoupons = async () => {
      try {
        const { data, error } = await supabase
          .from('coupons')
          .select('*')
          .order('id', { ascending: true })

        if (error) {
          console.error('Customer coupons error:', error)
          return
        }

        setCoupons((data ?? []).map(normalizeCoupon))
      } catch (error) {
        console.error('Customer coupons load failed:', error)
      }
    }

    loadProducts()
    loadCoupons()
  }, [])

  const categories = useMemo(
    () => [
      'الكل',
      ...Array.from(
        new Set(
          products
            .map((product) => product.category)
            .filter((category) => category && category.trim()),
        ),
      ),
    ],
    [products],
  )

  const filteredProducts =
    selectedCategory === 'الكل'
      ? products
      : products.filter(
          (product) => product.category === selectedCategory,
        )

  const subtotal = cart.reduce(
    (sum, item) => sum + item.price * item.quantity,
    0,
  )

  const discountAmount = subtotal * (discountPercent / 100)
  const total = Math.max(subtotal - discountAmount, 0)

  const addToCart = (product: Product) => {
    if (!product.available || product.stock <= 0) {
      notify('هذا المنتج غير متوفر حاليًا', 'error')
      return
    }

    setCart((currentCart) => {
      const existing = currentCart.find(
        (item) => item.id === product.id,
      )

      if (existing) {
        if (existing.quantity >= product.stock) {
          notify('لا توجد كمية كافية من هذا المنتج', 'error')
          return currentCart
        }

        return currentCart.map((item) =>
          item.id === product.id
            ? { ...item, quantity: item.quantity + 1 }
            : item,
        )
      }

      return [
        ...currentCart,
        {
          ...product,
          quantity: 1,
        },
      ]
    })

    setSelectedProduct(product)
  }

  const updateCartItemQuantity = (id: number, delta: number) => {
    setCart((currentCart) =>
      currentCart
        .map((item) => {
          if (item.id !== id) return item

          const nextQuantity = item.quantity + delta
          return {
            ...item,
            quantity: nextQuantity,
          }
        })
        .filter((item) => item.quantity > 0),
    )
  }

  const removeCartItem = (id: number) => {
    setCart((currentCart) =>
      currentCart.filter((item) => item.id !== id),
    )
  }

  const applyDiscount = async () => {
    const code = discountCode.trim().toUpperCase()

    if (!code) {
      setDiscountPercent(0)
      notify('اكتب كود الخصم أولاً', 'error')
      return
    }

    const localMatch = coupons.find(
      (coupon) =>
        coupon.code === code &&
        coupon.active !== false,
    )

    if (localMatch) {
      setDiscountPercent(localMatch.percent)
      notify(`تم تطبيق خصم ${localMatch.percent}%`, 'success')
      return
    }

    try {
      const { data, error } = await supabase
        .from('coupons')
        .select('*')
        .eq('code', code)
        .maybeSingle()

      if (error) {
        console.error('Coupon lookup failed:', error)
      }

      if (data) {
        const activeCoupon = normalizeCoupon(data)

        if (activeCoupon.active === false) {
          setDiscountPercent(0)
          notify('كود الخصم غير فعال', 'error')
          return
        }

        setDiscountPercent(activeCoupon.percent)
        notify(`تم تطبيق خصم ${activeCoupon.percent}%`, 'success')
        return
      }

      setDiscountPercent(0)
      notify('كود الخصم غير موجود أو غير فعال', 'error')
    } catch (error) {
      console.error('Coupon apply failed:', error)
      setDiscountPercent(0)
      notify('تعذّر التحقق من كود الخصم', 'error')
    }
  }

  const resetDiscount = () => {
    setDiscountCode('')
    setDiscountPercent(0)
  }

  const submitOrder = async () => {
    if (cart.length === 0) {
      notify('السلة فارغة', 'error')
      return
    }

    const normalizedPhone = phoneNumber.trim()
    if (!normalizedPhone) {
      notify('رقم الهاتف مطلوب', 'error')
      return
    }

    if (!orderType) {
      notify('يرجى اختيار نوع الطلب', 'error')
      return
    }

    if (orderType === 'داخل الكوفي' && !tableNumber.trim()) {
      notify('رقم الطاولة مطلوب عند الطلب داخل الكوفي', 'error')
      return
    }

    for (const item of cart) {
      const product = products.find(
        (entry) => entry.id === item.id,
      )

      if (!product || item.quantity > (product.stock ?? 0)) {
        notify(`الكمية غير كافية من ${item.name}`, 'error')
        return
      }
    }

    setIsSubmitting(true)

    const orderId = Date.now()
    const orderTimestamp = new Date().toISOString()
    const appliedCode =
      discountPercent > 0 ? discountCode.trim().toUpperCase() : ''

    const orderPayload = {
      id: orderId,
      items: cart.map((item) => ({
        id: item.id,
        name: item.name,
        category: item.category,
        price: item.price,
        quantity: item.quantity,
      })),
      type: orderType,
      table: orderType === 'داخل الكوفي' ? tableNumber : '',
      paymentMethod,
      discountCode: appliedCode,
      discountPercent,
      discountAmount,
      total,
      status: 'جديد',
      createdAt: new Date().toLocaleTimeString('ar-OM'),
      createdAtDate: orderTimestamp,
    }

    try {
      const { error } = await supabase
        .from('orders')
        .insert([orderPayload])

      if (error) {
        console.error('Customer order insert failed:', error)
        notify('تعذّر حفظ الطلب في Supabase', 'error')
        setIsSubmitting(false)
        return
      }

      const storedOrders = JSON.parse(
        localStorage.getItem('barni-orders') ?? '[]',
      )

      localStorage.setItem(
        'barni-orders',
        JSON.stringify([...storedOrders, orderPayload]),
      )

      for (const item of cart) {
        const currentProduct = products.find(
          (product) => product.id === item.id,
        )

        if (!currentProduct) continue

        const nextStock = Math.max(
          0,
          currentProduct.stock - item.quantity,
        )

        const { error: stockError } = await supabase
          .from('products')
          .update({
            stock: nextStock,
            active: nextStock > 0,
            available: nextStock > 0,
          })
          .eq('id', item.id)

        if (stockError) {
          console.error('Stock update failed:', stockError)
        }
      }

      setConfirmation({
        orderId,
        total,
        type: orderType,
        table: orderType === 'داخل الكوفي' ? tableNumber : 'سفري',
      })

      setCart([])
      setSelectedProduct(products[0] ?? null)
      setOrderType('')
      setPhoneNumber('')
      setTableNumber('')
      setDiscountCode('')
      setDiscountPercent(0)
      setPaymentMethod('نقدي')
      notify('تم إرسال الطلب بنجاح', 'success')
    } catch (error) {
      console.error('Order submission error:', error)
      notify('حدث خطأ أثناء تأكيد الطلب', 'error')
    } finally {
      setIsSubmitting(false)
    }
  }

  const totalItems = cart.reduce(
    (sum, item) => sum + item.quantity,
    0,
  )

  return (
    <div className="customer-page" dir="rtl">
      {showSplash && (
        <div className="barni-splash">
          <div className="barni-splash-logo">
            <span>برني</span>
          </div>
        </div>
      )}

      {!showSplash && showOrderChoice && (
        <div className="barni-order-choice">
          <div className="barni-choice-content">
            <div className="barni-choice-logo">برني</div>
            <div className="barni-choice-subtitle">كيف تحب تطلب؟</div>

            <div className="barni-choice-buttons">
              <button
                type="button"
                onClick={() => {
                  setOrderType('سفري')
                  setShowOrderChoice(false)
                }}
              >
                🚗 <strong>طلب من السيارة</strong>
              </button>

              <button
                type="button"
                onClick={() => {
                  setOrderType('داخل الكوفي')
                  setShowOrderChoice(false)
                }}
              >
                ☕ <strong>طلب من داخل المقهى</strong>
              </button>
            </div>
          </div>
        </div>
      )}

      {!showSplash && !showOrderChoice && (
        <div className="barni-app">
          <header className="barni-header">
            <div className="barni-header-inner">
              <button
                type="button"
                className="barni-simple-logo"
                onClick={() => setCustomerScreen('menu')}
              >
                برني
              </button>

              <div className="barni-header-actions">
                <button
                  type="button"
                  className="barni-order-mode"
                  onClick={() => setShowOrderChoice(true)}
                >
                  {orderType === 'داخل الكوفي' ? 'داخل المقهى' : 'من السيارة'}
                </button>

                <button
                  type="button"
                  className="barni-cart-button"
                  onClick={() => setCustomerScreen('cart')}
                >
                  السلة <b>{totalItems}</b>
                </button>
              </div>
            </div>
          </header>

          {customerScreen === 'menu' && (
            <main className="barni-page barni-menu-page">
              <div className="barni-menu-top">
                <h1>المنيو</h1>

                <div className="barni-category-tabs">
                  {categories.map((category) => (
                    <button
                      key={category}
                      type="button"
                      className={
                        selectedCategory === category
                          ? 'barni-category active'
                          : 'barni-category'
                      }
                      onClick={() => setSelectedCategory(category)}
                    >
                      {category}
                    </button>
                  ))}
                </div>
              </div>

              {products.length === 0 ? (
                <div className="barni-empty">
                  <span>☕</span>
                  <h3>لا توجد منتجات حاليًا</h3>
                </div>
              ) : (
                <div className="barni-product-grid">
                  {filteredProducts.map((product) => (
                    <article
                      key={product.id}
                      className="barni-product-card"
                      onClick={() => setSelectedProduct(product)}
                    >
                      <div className="barni-product-visual">
                        <div className="barni-product-ring" />
                        <span>☕</span>
                      </div>

                      <div className="barni-product-info">
                        <div>
                          <h3>{product.name}</h3>
                          <p>{product.category}</p>
                        </div>

                        <div className="barni-product-buy">
                          <strong>{product.price.toFixed(2)} ر.ع</strong>
                          <button
                            type="button"
                            disabled={!product.available || product.stock <= 0}
                            onClick={(event) => {
                              event.stopPropagation()
                              addToCart(product)
                            }}
                          >
                            +
                          </button>
                        </div>
                      </div>
                    </article>
                  ))}
                </div>
              )}

              {selectedProduct && (
                <div className="barni-product-modal" onClick={() => setSelectedProduct(null)}>
                  <div
                    className="barni-product-modal-card"
                    onClick={(event) => event.stopPropagation()}
                  >
                    <button
                      type="button"
                      className="barni-close"
                      onClick={() => setSelectedProduct(null)}
                    >
                      ×
                    </button>
                    <div className="barni-modal-image">☕</div>
                    <h2>{selectedProduct.name}</h2>
                    <p>{selectedProduct.category}</p>
                    <strong>{selectedProduct.price.toFixed(2)} ر.ع</strong>
                    <button
                      type="button"
                      className="barni-modal-add"
                      onClick={() => {
                        addToCart(selectedProduct)
                        setSelectedProduct(null)
                      }}
                      disabled={!selectedProduct.available || selectedProduct.stock <= 0}
                    >
                      إضافة للسلة
                    </button>
                  </div>
                </div>
              )}
            </main>
          )}

          {customerScreen === 'cart' && (
            <main className="barni-page barni-cart-page">
              <div className="barni-page-title">
                <button type="button" onClick={() => setCustomerScreen('menu')}>
                  ← المنيو
                </button>
                <h1>السلة</h1>
              </div>

              {cart.length === 0 ? (
                <div className="barni-cart-empty">
                  <span>🛒</span>
                  <h3>السلة فارغة</h3>
                  <button type="button" onClick={() => setCustomerScreen('menu')}>
                    العودة للمنيو
                  </button>
                </div>
              ) : (
                <div className="barni-cart-layout">
                  <section className="barni-cart-list">
                    {cart.map((item) => (
                      <div className="barni-cart-row" key={item.id}>
                        <div className="barni-cart-row-icon">☕</div>
                        <div className="barni-cart-row-main">
                          <h3>{item.name}</h3>
                          <span>{item.price.toFixed(2)} ر.ع</span>
                        </div>

                        <div className="barni-quantity">
                          <button type="button" onClick={() => updateCartItemQuantity(item.id, -1)}>−</button>
                          <strong>{item.quantity}</strong>
                          <button type="button" onClick={() => updateCartItemQuantity(item.id, 1)}>+</button>
                        </div>

                        <button
                          type="button"
                          className="barni-remove"
                          onClick={() => removeCartItem(item.id)}
                        >
                          حذف
                        </button>
                      </div>
                    ))}
                  </section>

                  <section className="barni-checkout-card">
                    <div className="barni-checkout-line">
                      <span>المجموع</span>
                      <strong>{subtotal.toFixed(2)} ر.ع</strong>
                    </div>

                    {discountPercent > 0 && (
                      <div className="barni-checkout-line discount">
                        <span>الخصم {discountPercent}%</span>
                        <strong>-{discountAmount.toFixed(2)} ر.ع</strong>
                      </div>
                    )}

                    <div className="barni-checkout-line total">
                      <span>الإجمالي</span>
                      <strong>{total.toFixed(2)} ر.ع</strong>
                    </div>

                    <label className="barni-field">
                      <span>رقم الهاتف</span>
                      <input
                        type="tel"
                        value={phoneNumber}
                        onChange={(event) => setPhoneNumber(event.target.value)}
                        placeholder="05xxxxxxxx"
                      />
                    </label>

                    {orderType === 'داخل الكوفي' && (
                      <label className="barni-field">
                        <span>رقم الطاولة</span>
                        <input
                          type="number"
                          value={tableNumber}
                          onChange={(event) => setTableNumber(event.target.value)}
                          placeholder="رقم الطاولة"
                        />
                      </label>
                    )}

                    <div className="barni-choice-group">
                      <span>طريقة الدفع</span>
                      <div>
                        {(['نقدي', 'بطاقة'] as const).map((method) => (
                          <button
                            key={method}
                            type="button"
                            className={paymentMethod === method ? 'active' : ''}
                            onClick={() => setPaymentMethod(method)}
                          >
                            {method}
                          </button>
                        ))}
                      </div>
                    </div>

                    <div className="barni-discount">
                      <input
                        type="text"
                        value={discountCode}
                        onChange={(event) => setDiscountCode(event.target.value.toUpperCase())}
                        placeholder="كود الخصم"
                      />
                      <button
                        type="button"
                        onClick={discountPercent > 0 ? resetDiscount : applyDiscount}
                      >
                        {discountPercent > 0 ? 'إزالة' : 'تطبيق'}
                      </button>
                    </div>

                    <button
                      type="button"
                      className="barni-submit"
                      onClick={submitOrder}
                      disabled={cart.length === 0 || isSubmitting}
                    >
                      {isSubmitting ? 'جاري الإرسال...' : 'تأكيد الطلب'}
                    </button>
                  </section>
                </div>
              )}

              {confirmation && (
                <div className="barni-confirmation">
                  <div className="barni-confirmation-icon">✓</div>
                  <h2>تم استلام طلبك</h2>
                  <p>رقم الطلب: #{confirmation.orderId}</p>
                  <p>
                    {confirmation.type === 'داخل الكوفي'
                      ? `طاولة ${confirmation.table}`
                      : 'طلب من السيارة'}
                  </p>
                  <strong>{confirmation.total.toFixed(2)} ر.ع</strong>
                </div>
              )}
            </main>
          )}

          {notice && <div className={`barni-toast ${notice.type}`}>{notice.message}</div>}
        </div>
      )}
    </div>
  )
}
